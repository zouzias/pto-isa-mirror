#
# pto-isa gated reduce-scatter ST — standalone executable.
#
# Output:
#   - gated_rs_st : pilot ST executable. Single-rank smoke test for the
#                   pto-native gated CCU kernel + AIV gate trigger pipeline.
#                   See `main.cc` for the call sequence and triage notes.
#
# Why standalone (not part of `tests/npu/a5/comm/st/testcase/`)
# ------------------------------------------------------------
# The pto-isa comm ST framework drives bisheng / `-xcce` because each ST kernel
# has a `__global__ AICORE` device kernel embedded in `<name>_kernel.cpp`. Our
# gated reduce-scatter ST has no AICORE kernel — the "device side" is CCU
# microcode generated by `Algorithm()` IR, and the AIV trigger is a pre-built
# `libpto_aiv_treduce.so`. So we use the host C++ toolchain (g++) and link
# the pre-built shared libs at runtime / link time.
#
# Build (server side):
#
#     # Required env or -D overrides:
#     #   ASCEND_HOME_PATH       : CANN install root (auto-detected if env set)
#     #   PTO_GATED_RS_LIB_DIR   : where libpto_gated_reduce_scatter.so lives
#     #   PTO_CCU_HOST_LIB_DIR   : where libpto_ccu_host.so lives
#     #   PTO_AIV_TREDUCE_LIB_DIR: where libpto_aiv_treduce.so lives
#
#     cmake -S tests/host/gated_reduce_scatter -B build/gated_rs_st \
#       -DPTO_GATED_RS_LIB_DIR=$PWD/build/pto_gated_reduce_scatter/dist \
#       -DPTO_CCU_HOST_LIB_DIR=$PWD/build/pto_ccu_host/dist \
#       -DPTO_AIV_TREDUCE_LIB_DIR=/path/to/aiv-treduce/dist
#     cmake --build build/gated_rs_st -j

cmake_minimum_required(VERSION 3.16)
project(gated_rs_st LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# ────────────────────────────────────────────────────────────────────────────
# pto-isa repo root for `<pto/...>` includes.
# ────────────────────────────────────────────────────────────────────────────
get_filename_component(PTO_LIB_PATH "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)

# ────────────────────────────────────────────────────────────────────────────
# CANN toolkit resolution (same defaults as `kernels/host/gated_reduce_scatter`).
# ────────────────────────────────────────────────────────────────────────────
if(NOT DEFINED ASCEND_HOME_PATH)
    if(DEFINED ENV{ASCEND_HOME_PATH})
        set(ASCEND_HOME_PATH $ENV{ASCEND_HOME_PATH})
    else()
        message(FATAL_ERROR "gated_rs_st: set ASCEND_HOME_PATH (env or -D).")
    endif()
endif()
message(STATUS "gated_rs_st: ASCEND_HOME_PATH = ${ASCEND_HOME_PATH}")

if(NOT DEFINED HCOMM_PKG_INC_DIR AND DEFINED ENV{HCOMM_PKG_INC_DIR})
    set(HCOMM_PKG_INC_DIR $ENV{HCOMM_PKG_INC_DIR})
endif()

if(NOT DEFINED HCOMM_PKG_INC_DIR)
    find_path(_HCOMM_PROBE
        NAMES hcomm/ccu/ccu_kernel.h
        PATHS
            ${ASCEND_HOME_PATH}/pkg_inc
            ${ASCEND_HOME_PATH}/include
            ${PTO_LIB_PATH}/../../hcomm/pkg_inc
            $ENV{HOME}/hcomm/pkg_inc
        NO_DEFAULT_PATH
    )
    if(NOT _HCOMM_PROBE)
        message(FATAL_ERROR
            "gated_rs_st: cannot locate hcomm/ccu/ccu_kernel.h. "
            "Set -DHCOMM_PKG_INC_DIR=<abs path> and re-run cmake.")
    endif()
    set(HCOMM_PKG_INC_DIR ${_HCOMM_PROBE})
endif()
message(STATUS "gated_rs_st: HCOMM_PKG_INC_DIR = ${HCOMM_PKG_INC_DIR}")

# ACL / hccl headers — same dual-root convention as
# `kernels/host/gated_reduce_scatter/CMakeLists.txt`.
if(NOT DEFINED CANN_INCLUDE_DIR AND IS_DIRECTORY "${ASCEND_HOME_PATH}/include")
    set(CANN_INCLUDE_DIR "${ASCEND_HOME_PATH}/include")
endif()

# Locate hcomm internal headers (`hcomm_primitives.h`, `hccl_types.h`,
# `hcomm_res_defs.h`) that pkg_inc transitively includes WITHOUT a prefix.
# See `kernels/host/gated_reduce_scatter/CMakeLists.txt` for the full
# rationale and probe-root rationale.
set(_st_internal_hdrs
    hcomm_primitives.h
    hccl_types.h
    hcomm_res_defs.h
)
set(_st_probe_roots
    ${HCOMM_PKG_INC_DIR}
    ${HCOMM_PKG_INC_DIR}/hcomm
    ${HCOMM_PKG_INC_DIR}/hccl
    ${CANN_INCLUDE_DIR}
    ${CANN_INCLUDE_DIR}/hcomm
    ${CANN_INCLUDE_DIR}/hccl
    ${ASCEND_HOME_PATH}/x86_64-linux/include
    ${ASCEND_HOME_PATH}/x86_64-linux/include/hcomm
    ${ASCEND_HOME_PATH}/x86_64-linux/include/hccl
    ${ASCEND_HOME_PATH}/aarch64-linux/include
    ${ASCEND_HOME_PATH}/aarch64-linux/include/hcomm
    ${ASCEND_HOME_PATH}/aarch64-linux/include/hccl
    ${PTO_LIB_PATH}/../../hcomm/include
    ${PTO_LIB_PATH}/../../hcomm/include/hcomm
    ${PTO_LIB_PATH}/../../hcomm/include/hccl
    $ENV{HOME}/hcomm/include
    $ENV{HOME}/hcomm/include/hcomm
    $ENV{HOME}/hcomm/include/hccl
)

set(HCOMM_INTERNAL_INC_DIRS "")
set(_missing_st_hdrs "")
foreach(_hdr ${_st_internal_hdrs})
    string(MAKE_C_IDENTIFIER "_ST_PROBE_${_hdr}" _probe_var)
    find_path(${_probe_var}
        NAMES ${_hdr}
        PATHS ${_st_probe_roots}
        NO_DEFAULT_PATH
    )
    if(${_probe_var})
        message(STATUS "gated_rs_st:   ${_hdr} -> ${${_probe_var}}")
        list(APPEND HCOMM_INTERNAL_INC_DIRS ${${_probe_var}})
    else()
        list(APPEND _missing_st_hdrs ${_hdr})
    endif()
endforeach()
if(_missing_st_hdrs)
    message(FATAL_ERROR
        "gated_rs_st: cannot locate hcomm internal header(s): "
        "${_missing_st_hdrs}\nSearched: ${_st_probe_roots}")
endif()
list(REMOVE_DUPLICATES HCOMM_INTERNAL_INC_DIRS)
message(STATUS
    "gated_rs_st: HCOMM_INTERNAL_INC_DIRS = ${HCOMM_INTERNAL_INC_DIRS}")

if(NOT DEFINED HCOMM_LIB_DIR)
    set(HCOMM_LIB_DIR "${ASCEND_HOME_PATH}/lib64")
endif()

# ────────────────────────────────────────────────────────────────────────────
# Sibling pto-isa lib locations.
# ────────────────────────────────────────────────────────────────────────────
foreach(_var PTO_GATED_RS_LIB_DIR PTO_CCU_HOST_LIB_DIR PTO_AIV_TREDUCE_LIB_DIR)
    if(NOT DEFINED ${_var})
        if(DEFINED ENV{${_var}})
            set(${_var} $ENV{${_var}})
        endif()
    endif()
    message(STATUS "gated_rs_st: ${_var} = ${${_var}}")
endforeach()

# ────────────────────────────────────────────────────────────────────────────
# Executable.
# ────────────────────────────────────────────────────────────────────────────
add_executable(gated_rs_st main.cc)

target_compile_options(gated_rs_st PRIVATE
    -Wall -Wextra -O2
)

target_include_directories(gated_rs_st PRIVATE
    ${PTO_LIB_PATH}/include
    ${HCOMM_PKG_INC_DIR}
    ${HCOMM_INTERNAL_INC_DIRS}
    # comm_mpi.h shim (dlopen-based MPI loader) — sibling pto-isa testcase
    # dir; we don't copy the file to keep a single source of truth. Required
    # by main.cc for multi-rank bootstrap (HcclCcuKernelRegister needs
    # nRanks >= 2 — see main.cc header).
    ${PTO_LIB_PATH}/tests/npu/a5/comm/st/testcase
)
if(DEFINED CANN_INCLUDE_DIR)
    target_include_directories(gated_rs_st PRIVATE ${CANN_INCLUDE_DIR})
endif()

# Link directories — one entry per shared lib supplier we depend on.
target_link_directories(gated_rs_st PRIVATE
    ${HCOMM_LIB_DIR}
)
if(DEFINED PTO_GATED_RS_LIB_DIR)
    target_link_directories(gated_rs_st PRIVATE ${PTO_GATED_RS_LIB_DIR})
endif()
if(DEFINED PTO_CCU_HOST_LIB_DIR)
    target_link_directories(gated_rs_st PRIVATE ${PTO_CCU_HOST_LIB_DIR})
endif()
if(DEFINED PTO_AIV_TREDUCE_LIB_DIR)
    target_link_directories(gated_rs_st PRIVATE ${PTO_AIV_TREDUCE_LIB_DIR})
endif()

# Link order:
#   - pto_gated_reduce_scatter : the gated kernel impl + factory
#                                (`MakeGatedKernelCreator`)
#   - pto_ccu_host             : Publish/TryGet registry. The kernel lib
#                                already links this transitively, but we add
#                                it explicitly so `pto::ccu::TryGet` resolves
#                                in the ST itself (we call it from main.cc).
#   - pto_aiv_treduce          : the AIV trigger lib. main.cc calls
#                                `pto::aiv::launch_treduce` directly.
#                                NOTE: `pto::aiv::launch_treduce` is
#                                `inline` per `pto/aiv/treduce.hpp`, so the
#                                actual symbol comes from the launcher impl
#                                in `kernels/manual/a5/treduce/launcher.cpp`,
#                                packaged as libpto_aiv_treduce.so. If the
#                                build host doesn't have this lib, link will
#                                fail with `undefined reference to launcher`
#                                — verify with `nm -D libpto_aiv_treduce.so`.
#   - hccl                     : HcclCommInitRootInfo, HcclCommDestroy, etc.
#   - hcomm                    : HcclCcuKernelRegister/Launch/Finish,
#                                HcclThreadAcquire, GetTokenInfo,
#                                QueryCcuBaseInfo (transitive: libhccp.so)
#   - ascendcl                 : ACL runtime (aclrtMalloc, aclInit, ...)
#   - dl, pthread              : transitive (registry mutex, pto::host probes)
target_link_libraries(gated_rs_st PRIVATE
    pto_gated_reduce_scatter
    pto_ccu_host
    pto_aiv_treduce
    hccl
    hcomm
    ascendcl
    dl
    pthread
)

# Embed RPATH so users don't have to set LD_LIBRARY_PATH for the sibling pto
# libs. CANN libs (hcomm/ascendcl/...) still need LD_LIBRARY_PATH or
# /etc/ld.so.conf.d/*.conf — same convention as existing pto-isa STs.
set(_rpath_list "${HCOMM_LIB_DIR}")
foreach(_var PTO_GATED_RS_LIB_DIR PTO_CCU_HOST_LIB_DIR PTO_AIV_TREDUCE_LIB_DIR)
    if(DEFINED ${_var})
        list(APPEND _rpath_list "${${_var}}")
    endif()
endforeach()
set_target_properties(gated_rs_st PROPERTIES
    INSTALL_RPATH         "${_rpath_list}"
    BUILD_WITH_INSTALL_RPATH ON
)

message(STATUS
    "gated_rs_st: executable will be built at ${CMAKE_BINARY_DIR}/gated_rs_st")
